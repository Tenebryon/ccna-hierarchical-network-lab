!
===================================================================================================================================================================================
! SW-DC-01 - Principais Funções
!
! Funcoes principais:
! - Switch de acesso do Data Center
! - Conexao do Windows Server através da VLAN 50
! - Redundancia de uplink com SW-CORE-01 e SW-CORE-02
! - VLAN 130 destinada ao gerenciamento
! - VLAN 333 utilizada como Native VLAN
!
===================================================================================================================================================================================
!
! Configurações Iniciais
!

en
conf t

hostname SW-DC-01
banner motd #! Somente Pessoas Autorizadas ! #

enable secret class

line console 0
 privilege level 15
 password class
 login
exit

line vty 0 4
 password class
 login
exit

service password-encryption

end
wr

!
===================================================================================================================================================================================
!
! Desativa a tentativa de resolucao DNS quando um comando incorreto e digitado.
!

en
conf t

no ip domain-lookup

end
wr

!
===================================================================================================================================================================================
!
! Criando as VLANs utilizadas nesse equipamento.
!

en
conf t

vlan 50
 name DC
exit

vlan 130
 name GER-DC-CORE
exit

vlan 333
 name Native
exit

end
wr

!
! Verificando as configurações aplicadas
!

sh vlan
sh vlan brief

!
===================================================================================================================================================================================
!
! Configurando Ethernet0/0 como porta de acesso da VLAN 50.
! Interface destinada ao Windows Server.
!

en
conf t

interface Ethernet0/0
 switchport mode access
 switchport access vlan 50
 no shutdown
end
wr

!
! Verificando as configurações aplicadas
!

sh int Ethernet0/0 switchport

!
===================================================================================================================================================================================
!
! Configurando os enlaces trunk para os switches CORE.
!
! Ethernet0/1 -> SW-CORE-01
! Ethernet0/3 -> SW-CORE-02
!
! VLAN 333 utilizada como Native VLAN.
! Permitindo o trafego das VLANs 50, 130.
!

en
conf t

interface range Ethernet0/1 , Ethernet0/3
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 333
 switchport trunk allowed vlan 50,130
 no shutdown
end
wr

!
! Verificando as configurações aplicadas
!

sh int switchport
sh int trunk

!
===================================================================================================================================================================================
!
! Configurando a interface VLAN de gerenciamento.
!

en
conf t

interface Vlan130
 description # ! Interface de Gerencia - 172.16.130.0/24 !
 ip address 172.16.130.12 255.255.255.0
 no shutdown
exit

ip default-gateway 172.16.130.1

end
wr

!
! Verificando as configurações aplicadas
!

sh run int vlan130
sh ip int brief | ex una

!
===================================================================================================================================================================================
!
