!
===================================================================================================================================================================================
! SW-LAB-05 - Principais Funções
!
! - Funcoes principais:
! - Switch de acesso do grupo LAB-SONS
! - Porta de acesso pertencente a VLAN 20
! - Enlaces trunk para transporte das VLANs
! - VLAN 120 destinada ao gerenciamento
! - VLAN 333 utilizada como Native VLAN nos enlaces trunk
!
===================================================================================================================================================================================
!
! Configurações Iniciais
!

en
conf t

hostname SW-LAB-05
banner motd #! Somente Pessoas Autorizadas ! #

enable secret class

line console 0
 privilege level 15
 password class
 login
exit

line vty 0 4
 password class
 login
exit

service password-encryption

end
wr

!
===================================================================================================================================================================================
!
! Desativa a tentativa de resolucao DNS quando um comando incorreto e digitado.
!

en
conf t

no ip domain-lookup

end
wr

!
===================================================================================================================================================================================
!
! Criando as VLANs utilizadas nesse equipamento.
!

en
conf t

vlan 20
 name LAB-SONS
exit

vlan 120
 name GER-AC-DST
exit

vlan 333
 name Native
exit

end
wr

!
! Verificando as configurações aplicadas
!

sh vlan
sh vlan brief

!
===================================================================================================================================================================================
!
! Configurando Ethernet0/0 como porta de acesso da VLAN 20.
!

en
conf t

interface Ethernet0/0
 switchport mode access
 switchport access vlan 20
 no shutdown
exit

end
wr

!
===================================================================================================================================================================================
!
! Configurando Ethernet0/2 e Ethernet0/3 como interfaces trunk.
! Permitindo o trafego das VLANs 20, 120 e configurando a vlan nativa.
!

en
conf t

interface range Ethernet0/2 , Ethernet0/3
 switchport trunk encapsulation dot1q
 switchport trunk allowed vlan 20,120
 switchport trunk native vlan 333
 switchport mode trunk
 no shutdown
end
wr

!
! Verificando as configurações aplicadas
!

sh int switchport
sh int trunk

!
===================================================================================================================================================================================
!
! Configurando a interface VLAN de gerenciamento.
!

en
conf t

interface Vlan120
 description # ! Interface de Gerencia - 172.16.120.0/24 !
 ip address 172.16.120.14 255.255.255.0
 no shutdown
exit

ip default-gateway 172.16.120.1

end
wr

!
! Verificando as configurações aplicadas
!

sh run int vlan120
sh ip int brief | ex una

!
===================================================================================================================================================================================
!
